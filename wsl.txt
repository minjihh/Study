# powershell에서 wsl 설치
wsl.exe --install
wsl --install -d Ubuntu-24.04 --name hermes11

# 설치된 버전 확인
wsl -l -v

# 설치할 수 있는 리눅스 버전확인
wsl -l -o

# 설치한거 실행
wsl.ext -d hermes11
wsl -d hermes11
